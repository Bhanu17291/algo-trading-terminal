    const clamp = (v, a, b) => Math.max(a, Math.min(b, v));
    const easeOutQuad = (t) => 1 - (1 - t) * (1 - t);

    function hexToRgba(hex, alpha){
      const h = String(hex).replace("#","").trim();
      const full = h.length===3 ? (h[0]+h[0]+h[1]+h[1]+h[2]+h[2]) : h;
      const n = parseInt(full, 16);
      const r = (n>>16)&255, g = (n>>8)&255, b = n&255;
      return "rgba(" + r + ", " + g + ", " + b + ", " + alpha + ")";
    }

    function applyGradientFromPalette(palette){
      const a = palette[0], b = palette[1], c = palette[2];
      document.documentElement.style.setProperty("--g1", a);
      document.documentElement.style.setProperty("--g2", b);
      document.documentElement.style.setProperty("--g3", c);
      document.documentElement.style.setProperty("--tint", hexToRgba(b, 0.18));
    }

    function el(tag, className, text){
      const node = document.createElement(tag);
      if (className) node.className = className;
      if (text !== undefined && text !== null) node.textContent = text;
      return node;
    }

    function buildCard(item, idx, activateFn, nextFn, prevFn){
      const card = el("article", "news-card");
      card.dataset.idx = String(idx);

      const collapsed = el("div", "collapsed");
      collapsed.appendChild(el("div", "mini-title", item.title));
      card.appendChild(collapsed);

      const expanded = el("div", "expanded");
      expanded.setAttribute("aria-hidden", "true");

      const left = el("div", "exp-left");
      const pill = el("span", "category-pill");
      pill.appendChild(el("span", "dot"));
      const strong = document.createElement("strong");
      strong.textContent = item.category;
      pill.appendChild(strong);
      left.appendChild(pill);
      left.appendChild(el("h2", "headline", item.title));
      left.appendChild(el("p", "subhead", item.subhead || ""));
      expanded.appendChild(left);

      const right = el("div", "exp-right");
      const rightTop = el("div", "right-top");

      const metaRow = el("div", "meta-row");
      metaRow.appendChild(el("span", "meta-chip", "⏱️ " + item.time));
      metaRow.appendChild(el("span", "meta-chip", "📍 " + item.place));
      metaRow.appendChild(el("span", "meta-chip", "🕒 " + item.updated));
      rightTop.appendChild(metaRow);

      rightTop.appendChild(el("p", "desc", item.desc));

      rightTop.appendChild(el("div", "section-title", "Key Highlights"));
      const ul = el("ul", "highlights");
      (item.highlights || []).forEach(h => ul.appendChild(el("li", "", h)));
      rightTop.appendChild(ul);

      const tagsWrap = el("div", "tags");
      (item.tags || []).forEach(t => tagsWrap.appendChild(el("span", "tag", "#" + t)));
      rightTop.appendChild(tagsWrap);

      rightTop.appendChild(el("p", "source", item.source || ""));
      right.appendChild(rightTop);

      const actions = el("div", "actions");
      const prevBtn = el("button", "btn", "← Prev");
      prevBtn.type = "button";
      prevBtn.dataset.action = "prev";
      const nextBtn = el("button", "btn", "Next →");
      nextBtn.type = "button";
      nextBtn.dataset.action = "next";
      actions.appendChild(prevBtn);
      actions.appendChild(nextBtn);
      right.appendChild(actions);

      expanded.appendChild(right);
      card.appendChild(expanded);

      const edgePrev = el("button", "edge-nav edge-prev");
      edgePrev.type = "button";
      edgePrev.dataset.action = "prev";
      edgePrev.setAttribute("aria-label", "Previous story");
      edgePrev.appendChild(el("span", "", "‹"));

      const edgeNext = el("button", "edge-nav edge-next");
      edgeNext.type = "button";
      edgeNext.dataset.action = "next";
      edgeNext.setAttribute("aria-label", "Next story");
      edgeNext.appendChild(el("span", "", "›"));

      card.appendChild(edgePrev);
      card.appendChild(edgeNext);

      card.addEventListener("click", () => activateFn(idx, true));
      card.addEventListener("click", (e) => {
        const btn = e.target.closest("[data-action]");
        if(!btn) return;
        e.stopPropagation();

      return card;
    }

    const cardEls = [];
    let active = 0;
    let timer = null;

    /**
     * RULES enforced:
     * 1) No small card above expanded card => y positions are within activeTop..activeBottom.
     * 2) Equal L/R cards => 7 items => 6 inactive => 3 left + 3 right.
     * 3) Size readable => globalScale min 0.90, and we scale all cards consistently.
     */
    function layout(activeIndex, instant){
      const stageRect = stageEl.getBoundingClientRect();
      const headerRect = headerEl.getBoundingClientRect();

      const W = stageRect.width;
      const H = stageRect.height;

      const padX = 18;
      const gapX = (W < 900) ? 18 : 26;
      const gapY = (W < 900) ? 14 : 18;

      const topSafe = (headerRect.bottom - stageRect.top) + 18;
      const bottomSafe = H - 70;

      const baseSmallW = (W <= 520) ? 210 : (W <= 980 ? 240 : 260);
      const baseSmallH = (W <= 520) ? 100 : (W <= 980 ? 112 : 120);

      const baseActiveW = Math.min(780, W * (W <= 820 ? 0.94 : 0.92));
      const baseActiveH = Math.min(520, H - (W <= 980 ? 220 : 230));

      const n = cardEls.length;
      const inactiveCount = n - 1;

      const leftCount = Math.floor(inactiveCount / 2);
      const rightCount = Math.floor(inactiveCount / 2);

      // active height must accommodate side stacks without pushing any above
      const requiredSideHeight = (leftCount * baseSmallH) + ((leftCount - 1) * gapY);

      const maxSafeH = Math.max(320, (bottomSafe - topSafe) - 8);
      const activeH = clamp(Math.max(baseActiveH, requiredSideHeight + 16), 340, maxSafeH);

      // horizontal fit; keep readable min scale
      const neededWidth = baseActiveW + 2*(baseSmallW + gapX) + 2*padX;
      const globalScale = clamp(W / neededWidth, 0.90, 1);

      const activeW = baseActiveW * globalScale;
      const smallW = baseSmallW * globalScale;
      const smallH = baseSmallH * globalScale;

      const cx = W / 2;
      let cy = clamp((topSafe + bottomSafe)/2,
        topSafe + activeH/2 + 8,
        bottomSafe - activeH/2 - 8
      );

      const activeTop = cy - activeH/2;
      const activeBot = cy + activeH/2;

      const inactive = [];
      for(let i=0;i<n;i++) if(i !== activeIndex) inactive.push(i);

      const leftIdxs = inactive.slice(0, leftCount);
      const rightIdxs = inactive.slice(leftCount, leftCount + rightCount);

      // apply active class + aria
      for(let i=0;i<n;i++){
        const c = cardEls[i];
        c.classList.toggle("active", i === activeIndex);
        const expanded = c.querySelector(".expanded");
        if (expanded) expanded.setAttribute("aria-hidden", i === activeIndex ? "false" : "true");
      }

      // place active
      const activeEl = cardEls[activeIndex];
      activeEl.style.setProperty("--x", cx + "px");
      activeEl.style.setProperty("--y", cy + "px");
      activeEl.style.setProperty("--s", globalScale.toFixed(3));
      activeEl.style.setProperty("--r", "0deg");
      activeEl.style.opacity = "1";
      activeEl.style.zIndex = "1000";

      // side x positions (guaranteed no overlap with active)
      const baseLeftX  = cx - activeW/2 - smallW/2 - gapX;
      const baseRightX = cx + activeW/2 + smallW/2 + gapX;

      // safe outward curve amplitude
      const leftRoom = (baseLeftX - smallW/2) - padX;
      const rightRoom = (W - padX) - (baseRightX + smallW/2);
      const curveAmp = clamp(Math.min(leftRoom, rightRoom, 44), 0, 44);

      // RULE #1: y range strictly within active span
      const yMin = activeTop + smallH/2;
      const yMax = activeBot - smallH/2;

      function yPositions(count){
        if(count <= 1) return [ (yMin + yMax) / 2 ];
        const out = [];
        for(let k=0;k<count;k++){
          const t = k/(count-1);
          out.push(yMin + (yMax - yMin) * t);
        }
        return out;
      }

      const leftYs = yPositions(leftCount);
      const rightYs = yPositions(rightCount);

      const spanHalf = Math.max(1, (yMax - yMin)/2);
      const centerY = (yMin + yMax)/2;

      function placeSideCard(cardIndex, side, y){
        const c = cardEls[cardIndex];

        const t = clamp(Math.abs(y - centerY)/spanHalf, 0, 1);
        const bend = easeOutQuad(t);

        // curve OUTWARD only
        const x = (side === "left")
          ? (baseLeftX - curveAmp * bend)
          : (baseRightX + curveAmp * bend);

        // subtle tilt for arc feel
        const tilt = (side === "left" ? -1 : 1) * (bend * 4.5);

        c.style.setProperty("--x", x.toFixed(2) + "px");
        c.style.setProperty("--y", y.toFixed(2) + "px");
        c.style.setProperty("--s", globalScale.toFixed(3));
        c.style.setProperty("--r", tilt.toFixed(2) + "deg");
        c.style.opacity = "0.92";
        c.style.zIndex = "200";
      }

      leftIdxs.forEach((idx, i) => placeSideCard(idx, "left", leftYs[i]));
      rightIdxs.forEach((idx, i) => placeSideCard(idx, "right", rightYs[i]));

      if(instant){
        for (let i=0;i<cardEls.length;i++) cardEls[i].style.transition = "none";
        requestAnimationFrame(() => {
          for (let i=0;i<cardEls.length;i++) cardEls[i].style.transition = "";
        });
      }
    }

    function restartAuto(){
      if(timer) clearTimeout(timer);
      const tick = () => {
        next(false);
        timer = setTimeout(tick, ROTATE_MS);
      };
      timer = setTimeout(tick, ROTATE_MS);
    }

    function activateFn(idx, userInitiated){
      active = ((idx % news.length) + news.length) % news.length;
      applyGradientFromPalette(news[active].palette);
      layout(active, false);
      if(userInitiated) restartAuto();
    }

    function next(userInitiated){ activateFn(active + 1, userInitiated); }
    function prev(userInitiated){ activateFn(active - 1, userInitiated); }

    // Build cards
    news.forEach((item, idx) => {
      const card = buildCard(item, idx, activateFn, next, prev);
      arcEl.appendChild(card);
      cardEls.push(card);
    });

    function runLoader(){
      let pct = 0;
      const stages = [
        { at: 20, msg: "Loading headlines" },
        { at: 45, msg: "Building cards" },
        { at: 70, msg: "Setting layout" },
        { at: 92, msg: "Final touches" }
      ];

      const t = setInterval(() => {
        pct = Math.min(99, pct + Math.ceil(Math.random()*6));
        loaderPct.textContent = pct + "%";
        loaderBar.style.width = pct + "%";
        for (let i=0;i<stages.length;i++){
          if (pct >= stages[i].at) loaderMsg.textContent = stages[i].msg;
        }
      }, 180);

      window.addEventListener("load", () => {
        clearInterval(t);
        requestAnimationFrame(() => {
          loaderMsg.textContent = "Ready";
          loaderPct.textContent = "100%";
          loaderBar.style.width = "100%";
          setTimeout(() => loaderEl.classList.add("hide"), 450);
        });
      });
    }

    window.addEventListener("resize", () => layout(active, true));

    runLoader();
    activateFn(0, false);
    layout(0, true);
    restartAuto();
  </script>
